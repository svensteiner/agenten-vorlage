 from intelligent_features.llm_responder import LLMResponder
+import pytest
+

+@pytest.mark.parametrize("task_type,max_tokens", [
+    ("source_faithfulness_review", 1200),
+    ("source_faithfulness_repair", 150),
+])
+def test_source_qualification_prompts_reach_transport_without_tweet_wrapper(task_type, max_tokens):
+    responder = LLMResponder.__new__(LLMResponder)
+    responder.config = {}
+    captured = {}
+
+    def call(messages, temperature, max_tokens, label, prefer_fast=False):
+        captured.update(messages=messages, temperature=temperature, max_tokens=max_tokens, label=label)
+        return '{"verdict":"reject"}'
+
+    responder._call_llm_api = call
+    user = 'SOURCE_JSON: {"observation":"Payment status unknown."}\nDRAFT_JSON: {"text":"Unpaid?"}'
+    system = "Return the exact requested review JSON or repair text; preserve draft language and limits."
+    result = responder._generate_openai_freeform(
+        user, system, {"task_type": task_type, "temperature": 0, "max_tokens": max_tokens},
+    )
+    assert result == '{"verdict":"reject"}'
+    assert captured["messages"] == [
+        {"role": "system", "content": system}, {"role": "user", "content": user},
+    ]
+    assert captured["temperature"] == 0
+    assert captured["max_tokens"] == max_tokens
+    assert captured["label"] == f"freeform/{task_type}"
+
