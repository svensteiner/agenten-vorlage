def require(condition: bool, message: str = "Protocol trace precondition failed") -> None:
    if not condition:
        raise ValueError(message)


def digest(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def read_json(path: Path):
    def unique(pairs):
        result = {}
        for key, value in pairs:
            if key in result:
                raise ValueError("duplicate JSON key")
            result[key] = value
        return result
    return json.loads(path.read_text(encoding="utf-8"), object_pairs_hook=unique)


def write_once(path: Path, value) -> None:
    payload = json.dumps(value, ensure_ascii=False, indent=2, allow_nan=False) + "\n"
    fd = os.open(path, os.O_CREAT | os.O_EXCL | os.O_WRONLY, 0o600)
    with os.fdopen(fd, "w", encoding="utf-8", newline="\n") as handle:
        handle.write(payload)
        handle.flush()
        os.fsync(handle.fileno())


def save_state(path: Path, value) -> None:
    payload = json.dumps(value, ensure_ascii=False, indent=2, allow_nan=False) + "\n"
    temporary = path.with_name(path.name + ".tmp")
    fd = os.open(temporary, os.O_CREAT | os.O_EXCL | os.O_WRONLY, 0o600)
    with os.fdopen(fd, "w", encoding="utf-8", newline="\n") as handle:
        handle.write(payload)
        handle.flush()
        os.fsync(handle.fileno())
    os.replace(temporary, path)


def load_parent():
    require(digest(PARENT_JOB) == PARENT_SHA256, "Parent transport helper hash changed")
    require(digest(V2_JOB) == V2_SHA256, "Pinned v2 synthetic diagnostic changed")
    require(digest(V3_JOB) == V3_SHA256, "Pinned v3 strict-review diagnostic changed")
    spec = importlib.util.spec_from_file_location("source_faithfulness_paired_parent", PARENT_JOB)
    if spec is None or spec.loader is None:
        raise ImportError("Pinned parent helper could not be loaded")
    module = importlib.util.module_from_spec(spec)
    sys.modules[spec.name] = module
    spec.loader.exec_module(module)
    # This is an explicit process-local override. The frozen paired helper is
    # never edited; only this new diagnostic process points its loader at the
    # post-wire-fix host component and matching source pin.
    module.SOURCES["source_faithfulness"] = module.WORKSPACE / "content/source_faithfulness.py"
    module.EXPECTED_SOURCE_SHA256["source_faithfulness"] = SOURCE_FAITHFULNESS_SHA256
    module.VALIDATION = FULL_GATE
    module.VALIDATION_LOG = FULL_LOG
    require(digest(module.SOURCES["source_faithfulness"]) == SOURCE_FAITHFULNESS_SHA256,
            "Post-wire-fix qualifier component hash changed")
    return module

class PayloadReceiptSession:
    """Inject strict schema for reviews and persist a credential-free payload receipt before HTTP."""

    def __init__(self, transport, state: dict, persist, post):
        self.transport = transport
        self.state = state
        self.persist = persist
        self.post = post

    def __call__(self, url, **kwargs):
        if self.transport.halted:
            raise RuntimeError("Transport halted; no schema-free retry")
        payload = kwargs.get("json")
        requests = self.state.get("requests")
        item = requests[-1] if isinstance(requests, list) and requests else None
        if (not isinstance(payload, dict) or not isinstance(item, dict)
                or item.get("status") != "started" or item.get("purpose") not in {"review", "repair"}):
            self.transport.halted = True
            raise RuntimeError("Unexpected outbound payload; halted")
        payload = copy.deepcopy(payload)
        purpose = item["purpose"]
        if purpose == "review":
            if "response_format" in payload:
                self.transport.halted = True
                raise RuntimeError("Unexpected preexisting response format; halted")
            payload["response_format"] = copy.deepcopy(RESPONSE_FORMAT)
            payload["provider"] = copy.deepcopy(REVIEW_PROVIDER_POLICY)
        else:
            if "response_format" in payload or payload.get("provider") != {"allow_fallbacks": False}:
                self.transport.halted = True
                raise RuntimeError("Unexpected preexisting repair format; halted")
            payload["response_format"] = copy.deepcopy(REPAIR_RESPONSE_FORMAT)
            payload["provider"] = copy.deepcopy(REPAIR_PROVIDER_POLICY)
        payload_sha = hashlib.sha256(json.dumps(
            payload, ensure_ascii=False, sort_keys=True, separators=(",", ":")
        ).encode("utf-8")).hexdigest()
        receipt = {"request_number": item.get("number"), "purpose": purpose,
                   "payload_sha256": payload_sha,
                   "response_format": copy.deepcopy(RESPONSE_FORMAT if purpose == "review"
                                                     else REPAIR_RESPONSE_FORMAT),
                   "provider": copy.deepcopy(payload.get("provider"))}
        self.state.setdefault("payload_receipts", []).append(receipt)
        item["transmitted_payload_sha256"] = payload_sha
        item["structured_output"] = True
        try:
            self.persist()
        except Exception:
            self.transport.halted = True
            item["status"] = "payload_receipt_failed_before_request"
            raise RuntimeError("Payload receipt persistence failed; halted before HTTP") from None
        kwargs["json"] = payload
        return self.post(url, **kwargs)


class TraceResponder:
    """Capture exact validated callback returns before forwarding to qualifier."""

    def __init__(self, transport, state: dict, persist):
        self.transport = transport
        self.state = state
        self.persist = persist

    @staticmethod
    def _sha(value: str) -> str:
        return hashlib.sha256(value.encode("utf-8")).hexdigest()

    def _append(self, event: dict) -> None:
        self.state["callback_trace"].append(event)
        try:
            self.persist()
        except Exception:
            self.transport.halted = True
            raise RuntimeError("Trace persistence failed; transport halted") from None

    def generate_freeform(self, user_prompt, system_prompt=None, context=None):
        if self.transport.halted:
            raise RuntimeError("Trace transport halted; no retry")
        context_type = (context.get("task_type") if isinstance(context, dict) else None) or "unspecified"
        request_count_before = len(self.state["requests"])
        metadata = {
            "case_id": self.transport.active_case,
            "context_type": context_type,
            "context": {key: value for key, value in (context or {}).items()
                        if key in {"temperature", "max_tokens", "task_type"}} if isinstance(context, dict) else None,
            "system_chars": len(system_prompt) if isinstance(system_prompt, str) else None,
            "user_chars": len(user_prompt) if isinstance(user_prompt, str) else None,
            "system_prompt_sha256": self._sha(system_prompt) if isinstance(system_prompt, str) else None,
            "user_prompt_sha256": self._sha(user_prompt) if isinstance(user_prompt, str) else None,
        }
        try:
            returned = self.transport.generate_freeform(
                user_prompt=user_prompt, system_prompt=system_prompt, context=context)
        except Exception:
            latest = self.state["requests"][-1] if len(self.state["requests"]) > request_count_before else None
            self._append({**metadata, "status": "transport_error_or_blocked",
                          "failure_cause": "unknown",
                          "request_number": latest.get("number") if latest else None,
                          "transport_status": latest.get("status") if latest else None,
                          "response_id": latest.get("response_id") if latest else None,
                          "raw_text": None})
            raise
        if not isinstance(returned, str) or len(returned) > 4000:
            self.transport.halted = True
            self._append({**metadata, "status": "invalid_return_type_or_length", "raw_text": None})
            raise RuntimeError("Trace callback return invalid; transport halted")
        latest = self.state["requests"][-1] if len(self.state["requests"]) > request_count_before else None
        event = {**metadata, "status": "returned", "raw_text": returned,
                 "returned_chars": len(returned), "returned_sha256": self._sha(returned),
                 "request_number": latest.get("number") if latest else None,
                 "transport_status": latest.get("status") if latest else None}
        self._append(event)
        return returned


def _unique_object_pairs(pairs):
    result = {}
    for key, value in pairs:
        if key in result:
            raise ValueError("duplicate JSON key")
        result[key] = value
    return result


def decode_repair_envelope(raw: object, max_chars: int = 190) -> str:
    """Decode only the declared strict repair object; never extract or truncate text."""
    if not isinstance(raw, str) or len(raw) > 4000:
        raise ValueError("invalid repair envelope")
    try:
        envelope = json.loads(raw, object_pairs_hook=_unique_object_pairs,
                              parse_constant=lambda _: (_ for _ in ()).throw(ValueError("nonstandard JSON")))
    except (ValueError, TypeError, RecursionError):
        raise ValueError("invalid repair envelope") from None
    if not isinstance(envelope, dict) or set(envelope) != {"text"}:
        raise ValueError("invalid repair envelope")
    text = envelope["text"]
    if not isinstance(text, str) or not text.strip() or len(text) > max_chars:
        raise ValueError("invalid repair text")
    try:
        text.encode("utf-8")
    except UnicodeEncodeError:
        raise ValueError("invalid repair text") from None
    return text


class RepairEnvelopeResponder:
    """Expose decoded repair text while recording its exact provider wire body."""

    def __init__(self, traced, transport, state: dict, persist, max_chars: int = 190):
        self.traced = traced
        self.transport = transport
        self.state = state
        self.persist = persist
        self.max_chars = max_chars

    def generate_freeform(self, user_prompt, system_prompt=None, context=None):
        if self.transport.halted:
            raise RuntimeError("Transport halted; no repair replay")
        if not isinstance(context, dict) or context.get("task_type") != "source_faithfulness_repair":
            self.transport.halted = True
            raise RuntimeError("Unexpected repair decoder role; halted")
        wire = self.traced.generate_freeform(user_prompt=user_prompt,
                                             system_prompt=system_prompt, context=context)
        wire_sha = hashlib.sha256(wire.encode("utf-8")).hexdigest()
        try:
            decoded = decode_repair_envelope(wire, self.max_chars)
            record = {"case_id": self.transport.active_case, "status": "decoded",
                      "wire_body": wire, "wire_body_sha256": wire_sha,
                      "decoded_text": decoded,
                      "decoded_text_sha256": hashlib.sha256(decoded.encode("utf-8")).hexdigest(),
                      "response_format": copy.deepcopy(REPAIR_RESPONSE_FORMAT),
                      "max_chars": self.max_chars}
        except ValueError:
            record = {"case_id": self.transport.active_case, "status": "rejected",
                      "wire_body": wire, "wire_body_sha256": wire_sha,
                      "decoded_text": None, "decoded_text_sha256": None,
                      "response_format": copy.deepcopy(REPAIR_RESPONSE_FORMAT),
                      "max_chars": self.max_chars, "reason_code": "invalid_repair_envelope"}
            self.state.setdefault("repair_decodes", []).append(record)
            self.transport.halted = True
            try:
                self.persist()
            except Exception:
                raise RuntimeError("Repair decode receipt persistence failed; halted") from None
            raise RuntimeError("Repair response did not match declared JSON contract") from None
        self.state.setdefault("repair_decodes", []).append(record)
        try:
            self.persist()
        except Exception:
            self.transport.halted = True
            raise RuntimeError("Repair decode receipt persistence failed; halted") from None
        return decoded


def run() -> None:
    if any(path.exists() for path in (RESULT, LOCK)):
        raise RuntimeError("Existing trace attempt must be inspected, never replayed")
    prepared = read_json(PREPARED)
    parent = load_parent()
    gate = validate_full_gate(parent)
    validate_prepared(prepared, parent, gate)

    fd = os.open(LOCK, os.O_CREAT | os.O_EXCL | os.O_WRONLY, 0o600)
    with os.fdopen(fd, "w", encoding="ascii") as handle:
        handle.write(str(os.getpid()))
        handle.flush()
        os.fsync(handle.fileno())
    state = {
        "study_id": prepared["study_id"],
        "scope": prepared["scope"],
        "status": "starting",
        "prepared_sha256": digest(PREPARED),
        "parent_helper_sha256": PARENT_SHA256,
        "v2_helper_sha256": V2_SHA256,
        "v3_helper_sha256": V3_SHA256,
        "structured_output": copy.deepcopy(prepared["structured_output"]),
        "repair_decoder": copy.deepcopy(prepared["repair_decoder"]),
        "result_contract": copy.deepcopy(prepared["result_contract"]),
        "component_sha256": prepared["component_sha256"],
        "full_suite_gate": gate,
        "physical_call_limit": MAX_CALLS,
        "local_reservation_limit_usd": str(MAX_USD),
        "records": [], "callback_trace": [], "requests": [], "payload_receipts": [],
        "repair_decodes": [],
        "reserved_usd": "0", "known_provider_cost_usd": "0",
        "unknown_outcome_requests": 0, "cost_cap_violated": False,
        "purpose_costs": {purpose: {"reserved_usd": "0", "known_provider_cost_usd": "0",
                                     "unknown_cost_calls": 0, "unknown_token_calls": 0}
                          for purpose in ("generation", "review", "repair")},
        "quality_result": "synthetic_protocol_diagnostic_not_quality_grade",
    }
    persist = lambda: save_state(RESULT, state)
    persist()
    key = None
    transport = None
    budget_module = None
    budget = None
    budget_token = None
    try:
        key = parent._load_key_only()
        _, faithfulness, budget_module = parent.load_components()
        parent.MAX_CALLS = MAX_CALLS
        parent.MAX_USD = MAX_USD
        parent.RESERVE_PER_CALL = RESERVE_PER_CALL
        budget, budget_token = parent._install_budget(budget_module)
        transport = parent.Transport(key, state, persist, budget_module, budget)
        transport.session.post = PayloadReceiptSession(
            transport, state, persist, transport.session.post)
        del key
        key = None
        traced = TraceResponder(transport, state, persist)
        repair_decoder = RepairEnvelopeResponder(traced, transport, state, persist, max_chars=190)
        state["status"] = "running"
        persist()
        for candidate in CANDIDATES:
            transport.active_case = candidate["case_id"]
            transport.phase = "qualification"
            try:
                qualified = faithfulness.qualify_source_draft(
                    SYNTHETIC_SOURCE, candidate["draft"], reviewer=traced,
                    repairer=repair_decoder, max_chars=190)
                record = {"case_id": candidate["case_id"], "draft": candidate["draft"],
                          "status": qualified.get("status") if isinstance(qualified, dict) else "invalid_result",
                          "verification": qualified.get("verification") if isinstance(qualified, dict) else None,
                          "review_calls": qualified.get("review_calls") if isinstance(qualified, dict) else None,
                          "repair_calls": qualified.get("repair_calls") if isinstance(qualified, dict) else None,
                          "reason_codes": qualified.get("reason_codes") if isinstance(qualified, dict) else None,
                          "qualified_text": qualified.get("text") if isinstance(qualified, dict) else None}
            except Exception:
                record = {"case_id": candidate["case_id"], "draft": candidate["draft"],
                          "status": "error_fail_closed"}
            state["records"].append(record)
            persist()
        state["status"] = "transport_stopped" if transport.halted else "diagnostics_complete"
    except Exception as error:
        state.update(status="stopped", error_type=type(error).__name__)
    finally:
        if transport is not None:
            transport.session.close()
        if budget is not None:
            state["task_budget"] = budget.summary()
        if budget_module is not None and budget_token is not None:
            budget_module._CURRENT_BUDGET.reset(budget_token)
        key = None
        persist()
    print(json.dumps({"status": state["status"], "calls": len(state["requests"]),
                      "known_provider_cost_usd": state["known_provider_cost_usd"],
                      "reserved_usd": state["reserved_usd"],
                      "quality_result": state["quality_result"]}, sort_keys=True))


if __name__ == "__main__":
    if len(sys.argv) != 2 or sys.argv[1] not in {"prepare", "run"}:
        raise SystemExit("Usage: python source_faithfulness_protocol_trace_v4_job.py prepare|run")
    {"prepare": prepare, "run": run}[sys.argv[1]]()
